ACTIVACIÓN DEL ENTORNO

# 1) crear el entorno
python -m venv .venv     # si falla, prueba: python3 -m venv .venv

# 2) activarlo (Linux en Codespaces)
source .venv/bin/activate

# 3) instalar dependencias del proyecto
pip install -r requirements.txt   # si pip no aparece, prueba: python -m pip install -r requirements.txt

# 4) verificar que estás usando el Python del venv
which python
# debe terminar en .../.venv/bin/python
